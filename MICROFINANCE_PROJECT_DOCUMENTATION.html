/Common/DataTable.jsx
import React from 'react';
import { Inbox } from 'lucide-react';

export const DataTable = ({ columns, data, onRowClick }) => {
  return (
    <div style={{ overflowX: 'auto', borderRadius: '12px', border: '1px solid rgba(234, 179, 8, 0.35)', background: '#060608' }}>
      <table style={{ width: '100%', borderCollapse: 'collapse', fontSize: '13.5px' }}>
        <thead>
          <tr style={{ background: '#0d0d10', borderBottom: '1px solid rgba(234, 179, 8, 0.35)' }}>
            {columns.map((col, idx) => (
              <th key={idx} style={{ padding: '12px 18px', color: '#facc15', fontSize: '12px' }}>{col.header}</th>
            ))}
          </tr>
        </thead>
        <tbody>
          {data && data.length > 0 ? (
            data.map((row, rIdx) => (
              <tr key={rIdx} onClick={() => onRowClick && onRowClick(row)}>
                {columns.map((col, cIdx) => (
                  <td key={cIdx} style={{ padding: '14px 18px', color: '#fef08a' }}>
                    {col.render ? col.render(row) : row[col.accessor]}
                  </td>
                ))}
              </tr>
            ))
          ) : (
            <tr><td colSpan={columns.length} style={{ padding: '48px', textAlign: 'center', color: '#ca8a04' }}>No records found</td></tr>
          )}
        </tbody>
      </table>
    </div>
  );
};
/Borrower/BorrowerRegistrationForm.jsx
import React, { useState } from 'react';
import { borrowerApi } from '../../api/borrowerApi';

export const BorrowerRegistrationForm = ({ onSuccess }) => {
  const [formData, setFormData] = useState({
    firstName: 'Sunita', lastName: 'Devi', mobileNumber: '9876543210',
    aadhaarNumber: '987654321012', village: 'Shivpur', annualHouseholdIncome: '180000'
  });

  const handleSubmit = async (e) => {
    e.preventDefault();
    await borrowerApi.register(formData);
    alert('Borrower registered successfully!');
    if (onSuccess) onSuccess();
  };

  return (
    <form onSubmit={handleSubmit}>
      <input className="form-input" value={formData.firstName} onChange={e => setFormData({...formData, firstName: e.target.value})} />
      <input className="form-input" value={formData.mobileNumber} onChange={e => setFormData({...formData, mobileNumber: e.target.value})} maxLength={10} />
      <button type="submit" className="btn-primary">Register Borrower</button>
    </form>
  );
};

api.js (Axios Client Configuration)
import axios from 'axios';

const api = axios.create({
  baseURL: '/api',
  headers: { 'Content-Type': 'application/json' }
});

api.interceptors.request.use((config) => {
  const token = localStorage.getItem('token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

export default api;
